```dataviewjs
// 1. 正确获取当前页面
const page = dv.current();

// 2. 读取 SPENDINGS 字段（它是一个数组）
const spendingsRaw = page.SPENDINGS;
// 1. 标准化：确保 spendingsList 始终是一个数组
let spendingsList = [];
if (spendingsRaw) {
    if (Array.isArray(spendingsRaw)) {
        spendingsList = spendingsRaw;          // 已经是数组，直接使用
    } else if (typeof spendingsRaw === 'string') {
        spendingsList = [spendingsRaw];        // 单个字符串，转为单元素数组
    } else {
        // 如果值是数字或其他类型，也可以转为字符串处理（可选）
        spendingsList = [String(spendingsRaw)];
    }
}

let total = 0;
// 3. 判断是否存在
if (!spendingsList || spendingsList.length === 0) {
    dv.paragraph("今日暂无支出记录");
} else {
    // 4. 遍历数组，提取数字并求和
    let details = [];
    
    spendingsList.forEach(item => {
        // 用正则提取数字（支持小数）
        const match = item.match(/(\d+\.?\d*)/);
        if (match) {
            const amount = parseFloat(match[0]);
            total += amount;
            details.push(`${item} → ${amount}元`);
        } else {
            details.push(`${item} → 无法识别金额`);
        }
    });
	total = Math.round(total * 100) / 100;
    // 5. 显示明细和总和
    dv.paragraph("📋 **支出明细：**");
    dv.list(details);
    dv.paragraph(`---\n💰 **今日总支出：${total} 元**`);
}

// 6. 获取当前文件的句柄 (TFile 对象)
//const currentFile = app.workspace.getActiveFile();
// 方式B：从 dv.current() 提取路径再转换（不用也完全可以）
const filePath = dv.current().file.path;
const currentFile = app.vault.getAbstractFileByPath(filePath);
// 3. 使用 processFrontMatter 安全地更新 Frontmatter
await app.fileManager.processFrontMatter(currentFile, (frontmatter) => {
    // 将 total 赋值给一个名为 "total_spendings" 的属性
    frontmatter["spendings"] = total;
});
```
